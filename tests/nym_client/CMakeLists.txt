cmake_minimum_required(VERSION 3.16)
project(nym_client_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Network)
add_executable(nym_client_tests manager_tests.cpp ../../src/main/NymClient.cpp ../../src/main/NymClient.h ../../src/main/NymAccountStore.cpp)
target_compile_definitions(nym_client_tests PRIVATE NYM_CLIENT_TEST)
target_link_libraries(nym_client_tests PRIVATE Qt6::Core Qt6::Network)
configure_file(fake_client.py xmr-nym-client COPYONLY)
file(CHMOD ${CMAKE_CURRENT_BINARY_DIR}/xmr-nym-client PERMISSIONS OWNER_READ OWNER_WRITE OWNER_EXECUTE)
enable_testing()
add_test(NAME nym_client COMMAND nym_client_tests)

add_executable(nym_store_tests store_tests.cpp ../../src/main/NymAccountStore.cpp)
target_compile_definitions(nym_store_tests PRIVATE NYM_CLIENT_TEST)
target_link_libraries(nym_store_tests PRIVATE Qt6::Core)
add_test(NAME nym_store COMMAND nym_store_tests)
